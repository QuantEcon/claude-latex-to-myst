\begin{algorithm}
    \DontPrintSemicolon
    input $\sigma_0 \in \Sigma$, set $k \leftarrow 0$ and $\epsilon \leftarrow 1$ \;
    \While{$\epsilon > 0 $}
    {
        $h_k \leftarrow $ the fixed point of $\hat T_{\sigma_k}$ \;
        $\sigma_{k+1} \leftarrow $ an $h_k$-greedy policy, satisfying
        %
        \begin{equation*}
            \sigma_{k+1}(w, e) \in \argmax_{w' \in \Gamma(w, e)}
            \left\{
                (1-\beta) r(w, w', e)^\alpha + \beta h_k(w')^\alpha
            \right\}^{1/\alpha}
        \end{equation*}
        %
        \;
        \vspace{-1em}
        $k \leftarrow k + 1$ \;
    }
    \Return{$\sigma_{k-1}$}
    \caption{\label{algo:hpowb} Solving via Howard policy iteration}
\end{algorithm}
