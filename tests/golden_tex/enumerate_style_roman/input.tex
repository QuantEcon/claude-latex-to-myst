\section{Parts}

Prove the following:

\begin{enumerate}
    \item $T$ is a contraction, and
    \item the fixed point is unique.
\end{enumerate}

\begin{Exercise}\label{ex:parts}
    Show that:
    \begin{enumerate}
        \item part one, and
        \item part two with a nested list:
        \begin{enumerate}
            \item inner alpha
            \item inner beta
        \end{enumerate}
    \end{enumerate}
\end{Exercise}
