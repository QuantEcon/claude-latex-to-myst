\section{Norms}

Recall the norm properties:

\begin{multicols}{2}
    \begin{enumerate}
        \item[(a)] $\| u \| \geq 0$
        \item[(b)] $\| u \| = 0 \iff u = 0$
        \item[(c)] $\| \alpha u \| = |\alpha| \| u \|$ and
        \item[(d)] $\| u + v \| \leq \| u \| + \| v \|$
        \item[] (nonnegativity)
        \item[] (positive definiteness)
        \item[] (absolute homogeneity)
        \item[] (triangle inequality)
    \end{enumerate}
\end{multicols}
